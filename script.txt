    <script>
        let cachedData = null;

        async function fetchMarketData() {
            try {
                const res = await fetch('http://127.0.0.1:8000/api/screen');
                cachedData = await res.json();
                
                document.getElementById('live-status').className = 'px-3 py-1 text-xs rounded-full bg-emerald-950 text-emerald-400 border border-emerald-800';
                document.getElementById('live-status').innerText = 'Connected (' + new Date().toLocaleTimeString() + ')';
                
                updateMarketUI(cachedData.market);
                renderTable();
            } catch (err) {
                document.getElementById('live-status').className = 'px-3 py-1 text-xs rounded-full bg-rose-950 text-rose-400 border border-rose-800';
                document.getElementById('live-status').innerText = 'Connection Error: Make sure uvicorn is running';
            }
        }

        function updateMarketUI(m) {
            document.getElementById('nifty-price').innerText = '₹' + m.price;
            document.getElementById('nifty-dma').innerText = m.dma50 + ' / ' + m.dma200;
            document.getElementById('nifty-rsi').innerText = m.rsi;
            
            const regimeEl = document.getElementById('nifty-regime');
            regimeEl.innerText = m.regime;
            if (m.regime.includes('AGGRESSIVE')) {
                regimeEl.className = 'text-xs font-bold px-2 py-1 rounded bg-emerald-900 text-emerald-300';
            } else if (m.regime.includes('CAUTION')) {
                regimeEl.className = 'text-xs font-bold px-2 py-1 rounded bg-yellow-900 text-yellow-300';
            } else {
                regimeEl.className = 'text-xs font-bold px-2 py-1 rounded bg-rose-900 text-rose-300';
            }
        }

        function renderTable() {
            if (!cachedData || !cachedData.stocks) return;
            const capital = parseFloat(document.getElementById('total-capital').value) || 100000;
            const riskPct = parseFloat(document.getElementById('risk-pct').value) || 1.0;
            const maxRiskAmount = capital * (riskPct / 100);

            const tbody = document.getElementById('screener-rows');
            tbody.innerHTML = '';

            cachedData.stocks.forEach(stock => {
                const slPrice = (stock.price * 0.95).toFixed(2);
                const riskPerShare = stock.price - slPrice;
                const maxQty = Math.floor(maxRiskAmount / riskPerShare);

                const tr = document.createElement('tr');
                tr.className = stock.eligible ? 'bg-gray-800 hover:bg-gray-750' : 'opacity-40 hover:opacity-100';

                tr.innerHTML = `
                    <td class="p-3 font-semibold text-white">${stock.symbol}</td>
                    <td class="p-3">₹${stock.price}</td>
                    <td class="p-3">
                        <span class="px-2 py-0.5 text-xs rounded ${stock.price > stock.dma200 ? 'bg-emerald-950 text-emerald-400' : 'bg-rose-950 text-rose-400'}">
                            ${stock.price > stock.dma200 ? 'Bullish' : 'Below 200 DMA'}
                        </span>
                    </td>
                    <td class="p-3 ${stock.rsi >= 55 && stock.rsi <= 70 ? 'text-emerald-400 font-bold' : ''}">${stock.rsi}</td>
                    <td class="p-3 ${stock.pct_50dma <= 10 ? 'text-emerald-400' : 'text-rose-400'}">+${stock.pct_50dma}%</td>
                    <td class="p-3">
                        ${stock.pattern !== 'None' 
                            ? `<span class="px-2 py-1 rounded bg-indigo-900 text-indigo-300 font-bold text-xs">${stock.pattern}</span>` 
                            : '<span class="text-gray-500">No Setup</span>'}
                    </td>
                    <td class="p-3 text-rose-400 font-mono">₹${slPrice}</td>
                    <td class="p-3 font-bold text-emerald-400 font-mono">${stock.eligible ? maxQty + ' shares' : '0'}</td>
                `;
                tbody.appendChild(tr);
            });
        }

        fetchMarketData();
        setInterval(fetchMarketData, 30000);
    </script>
